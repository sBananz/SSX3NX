#  =====================================================================
#   SSX 3 for Nintendo Switch (SSX3NX) -- settings
#  =====================================================================
#
#  One setting per line: NAME=value. A line starting with '#' is ignored,
#  so to turn a setting on, remove the '#' in front of it.
#
#  The values below are the tested defaults: stock Switch clocks, a steady
#  30 FPS in races, 60 FPS in the menus. You don't need to change anything.
#
#  Sections:
#    1. Controls
#    2. Picture
#    3. Performance and clocks
#    4. Docked / Switch Lite
#    5. Startup and storage
#    6. Troubleshooting (only if something looks or sounds wrong)
#    7. Developer tools (frame-rate display, profilers, logging)


#  ---------------------------------------------------------------------
#   1. CONTROLS
#  ---------------------------------------------------------------------

# Riding layout. 1 = PS2-style: B jump, Y boost / tweak, A handplant,
# L / R / ZL / ZR the four grabs, X (or -) reset.
# 0 = the Xbox layout by button position (A jump and confirm).
# Menus always use the Nintendo buttons: A confirm, B back.
SSX3NX_PS2_CONTROLS=1

# On-screen button prompts drawn as Switch buttons. 0 = the original Xbox icons.
#SSX3NX_BUTTON_ICONS=0

# Options > Controller and the loading screens show a Switch Pro Controller.
# 0 = the original Xbox controller.
#SSX3NX_PRO_CONTROLLER=0

# Your own controller picture: put pro_controller.png (PNG with alpha, 4:3,
# e.g. 880x656) in sdmc:/switch/ssx3nx/icons/.

# "Please don't turn off your Switch" on the saving screens. 0 = the Xbox wording.
#SSX3NX_SWITCH_TEXT=0


#  ---------------------------------------------------------------------
#   2. PICTURE
#  ---------------------------------------------------------------------

# Menus at 16:9. 1 = stretched evenly to widescreen, 0 = 4:3 with side bars
# as the game drew them, 2 = widescreen 3D with the menus kept 4:3.
RECOMP_WIDESCREEN_MENUS=1

# Boost look: speed lines and a soft blur at the screen edges while you
# boost. 2 = speed lines only, 0 = off.
SSX3NX_BOOST_LINES=1

# Smoothing and texture options (0 = off). These cost GPU time; the
# defaults are what holds 30 FPS.
RECOMP_MSAA=1
RECOMP_ANISO=0
RECOMP_HDTEX=0
RECOMP_CRT=0

# Internal resolution in lines (default 720 = the handheld screen).
#RECOMP_RES=720


#  ---------------------------------------------------------------------
#   3. PERFORMANCE AND CLOCKS
#  ---------------------------------------------------------------------

# Clock speeds in MHz while the game runs (0 = leave the system's).
# The stock clocks below hold 30 FPS. For extra headroom in the busiest
# moments (the start of Metro City), you can overclock to CPU 1785,
# GPU 768, memory 1600 -- at the cost of battery and heat.
SSX3NX_CPU_MHZ=1020
SSX3NX_GPU_MHZ=384
SSX3NX_MEM_MHZ=1331

# 1 = use the clocks above even while sys-clk / Horizon OC is running.
# 0 = let sys-clk's profile decide. (A .nro runs inside whichever game
# hosts the Homebrew Menu, so a sys-clk profile made for "SSX 3" never applies.)
SSX3NX_CLOCK_FORCE=1

# Draw distance (1.0 = the original game). 0.85 keeps races at 30 FPS
# with six riders on screen.
RECOMP_DRAW_DIST=0.85

# Dynamic draw distance: when a frame runs late, the view pulls in a little
# (down to RECOMP_DYN_DRAW_MIN x the original) and eases back out when
# there's time again. 0 = a fixed view.
#RECOMP_DYN_DRAW_DIST=0
RECOMP_DYN_DRAW_MIN=0.5

# Rider shadow detail (1 = the original 128x128; higher costs a lot).
RECOMP_SHADOW_RES=1

# Frame rate. Physics and controls always run at 60 Hz underneath.
# 3 = adaptive: races at a steady 30, going up to 60 while there is time
#     for it and dropping straight back to 30 when frames run late.
# 2 = races always at 30 (the game's own half-rate mode); menus at 60.
# 0 = try to render every update (not steady on the Switch).
#RECOMP_FPS30=3
# Adaptive tuning: game-thread ms per frame allowed for 60 (of 16.7),
# seconds of headroom before going up, missed-flip score that drops to 30.
#SSX3NX_FPS60_WORK=14
#SSX3NX_FPS60_HOLD=2
#SSX3NX_FPS60_MISS=1.5

#  ---------------------------------------------------------------------

# 720 for more headroom; 0 = same as handheld) and switches to the docked

#SSX3NX_DOCK_RES=1080
#SSX3NX_DOCK_CPU_MHZ=1020
#SSX3NX_DOCK_GPU_MHZ=768

# Switch Lite: it can't dock, so 1 makes the game act docked anyway
# (docked clocks, and the 1080 picture scaled down to its screen -- sharper,

#RECOMP_PB_RUN_BATCH=0

# Renderer: vk (Vulkan, the default) or sw (software -- very slow).

RECOMP_TEX_HASH_SKIP=0
